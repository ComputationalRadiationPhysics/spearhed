cmake_minimum_required(VERSION 3.28)
project(PMaccPackage LANGUAGES CXX)

option(PMACC_PACKAGE_BUILD_PMACC "Build PMacc" ON)
option(PMACC_PACKAGE_BUILD_CARAVAN "Build Caravan" ON)
option(PMACC_PACKAGE_BUILD_LLAMALITE "Build llamaLite" ON)
if(PMACC_BUILD_TESTING OR CARAVAN_BUILD_TESTING OR LLAMALITE_BUILD_TESTING)
    include(CTest)
    enable_testing()
endif()
if(TARGET pmacc AND NOT TARGET pmacc::pmacc)
    message(FATAL_ERROR "Existing pmacc target has no pmacc::pmacc alias")
endif()
if((TARGET caravan::mpi OR TARGET caravan::alpaka) AND NOT TARGET caravan::core)
    message(FATAL_ERROR "Caravan adapters exist without caravan::core")
endif()
set(_pmacc_build_here OFF)
if(PMACC_PACKAGE_BUILD_PMACC AND NOT TARGET pmacc::pmacc)
    set(_pmacc_build_here ON)
endif()
if(_pmacc_build_here AND NOT PMACC_PACKAGE_BUILD_CARAVAN AND (NOT TARGET caravan::mpi OR NOT TARGET caravan::alpaka))
    message(FATAL_ERROR "PMacc requires Caravan's MPI and alpaka adapters")
endif()
if(_pmacc_build_here AND TARGET caravan::core AND (NOT TARGET caravan::mpi OR NOT TARGET caravan::alpaka))
    message(FATAL_ERROR "Existing Caravan lacks MPI or alpaka adapter required by PMacc")
endif()
if(_pmacc_build_here AND NOT PMACC_PACKAGE_BUILD_LLAMALITE AND NOT TARGET llamaLite::llamaLite)
    find_package(llamaLite CONFIG REQUIRED)
endif()

if(_pmacc_build_here)
    include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Backend.cmake")
    if(NOT TARGET alpaka::alpaka)
        set(PMACC_ALPAKA_PROVIDER internal CACHE STRING "alpaka dependency source")
        if(PMACC_ALPAKA_PROVIDER STREQUAL "internal")
            add_subdirectory(thirdParty/alpaka EXCLUDE_FROM_ALL)
        elseif(PMACC_ALPAKA_PROVIDER STREQUAL "external")
            find_package(alpaka 3 CONFIG REQUIRED)
        else()
            message(FATAL_ERROR "PMACC_ALPAKA_PROVIDER must be internal or external")
        endif()
    endif()
    if(PMACC_BACKEND STREQUAL "GpuCuda" OR PMACC_BACKEND STREQUAL "GpuHip")
        if(NOT TARGET mallocMC::mallocMC)
            add_subdirectory(thirdParty/mallocMC EXCLUDE_FROM_ALL)
        endif()
    endif()
    if(PMACC_PACKAGE_BUILD_CARAVAN AND NOT TARGET caravan::core)
        if(
            (DEFINED CARAVAN_BUILD_MPI AND NOT CARAVAN_BUILD_MPI)
            OR (DEFINED CARAVAN_BUILD_ALPAKA AND NOT CARAVAN_BUILD_ALPAKA)
        )
            message(FATAL_ERROR "PMacc requires CARAVAN_BUILD_MPI=ON and CARAVAN_BUILD_ALPAKA=ON")
        endif()
        set(CARAVAN_BUILD_MPI ON)
        set(CARAVAN_BUILD_ALPAKA ON)
    endif()
endif()
if(PMACC_PACKAGE_BUILD_CARAVAN AND NOT TARGET caravan::core)
    add_subdirectory(libs/caravan)
endif()
if(PMACC_PACKAGE_BUILD_LLAMALITE AND NOT TARGET llamaLite::llamaLite)
    add_subdirectory(libs/llamaLite)
endif()
if(_pmacc_build_here)
    include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/PMaccLibrary.cmake")
endif()
