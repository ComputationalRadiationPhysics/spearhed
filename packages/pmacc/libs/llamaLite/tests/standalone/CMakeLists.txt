cmake_minimum_required(VERSION 3.25)
project(llamaLiteStandalone LANGUAGES CXX)

if(PROJECT_IS_TOP_LEVEL)
    enable_testing()
endif()

if(NOT TARGET llamaLite::llamaLite)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." llamaLite)
endif()
if(TARGET llamaLite)
    message(FATAL_ERROR "The legacy unnamespaced llamaLite target must not exist")
endif()
if(NOT TARGET llamaLite::llamaLite)
    message(FATAL_ERROR "Expected public target llamaLite::llamaLite")
endif()
add_executable(llamaLiteStandalone main.cpp)
target_link_libraries(llamaLiteStandalone PRIVATE llamaLite::llamaLite)
add_test(NAME llamaLite_standalone COMMAND llamaLiteStandalone)
